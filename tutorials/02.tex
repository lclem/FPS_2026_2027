\section*{Algebraic power series}

\begin{exercise}[Algebraic series and context-free languages]
    Determine the number of words of length $2n$ in the Dyck
    language, \ie the language of well-parenthesized words.

    \emph{Hint:} use the identity $\sqrt{1 - 4x} = 1 -
    \sum_{n=1}^{\infty} \frac{2}{n} \binom{2n-2}{n-1} x^n$ to avoid
    tedious calculations (this can be shown using the generalized
    binomial theorem).
\end{exercise}

\begin{exercise}[Compositional inverses]
    Let $f \in \powerseries \Q x$ be a power series.
    It admits a \emph{right compositional inverse} if there is $g \in \powerseries \Q x$ \st~$f(g(x)) = x$.
    %
    \begin{enumerate}
        \item When does $f$ admit a right compositional inverse?
        \item If $f$ admits a right compositional inverse, does it admit a left compositional inverse?
        \item If $f$ admits a right compositional inverse and is algebraic, is the inverse also algebraic?
    \end{enumerate}
\end{exercise}

\begin{exercise}[Lagrange inversion formula]
    The purpose of this exercise is to develop a purely algebraic proof of the Lagrange inversion formula.
    Let $\phi \in \powerseries \Q w$ be a power series with $\phi(0) \neq 0$
    and let $f \in \powerseries \Q x$ be the unique power series \st
    \begin{align*}
        f = x \phi(f).
    \end{align*}
    %
    Show that for all $n \geq 1$,

    \begin{align*}
        \coef {x^n} {f(x)} = \frac 1 n \coef {w^{n-1}} {\phi(w)^n}.
    \end{align*}
\end{exercise}

\begin{exercise}
    Let $T$ be unique power series \st~$T = x e^T$.
    Use the Lagrange inversion formula to show that
    $\coef {x^n} T = \frac {n^{n-1}} {n!}$.
\end{exercise}
