\section*{D-finite power series}

\begin{exercise}
    Consider power series $f_1, \dots, f_k \in \powerseries \Q x$ satisfying a system of linear differential equations over $\rat \Q x$
    %
    \begin{align}
        f_i' = A_{i1} \cdot f_1 + \dots + A_{ik} \cdot f_k,
        \quad A_{ij} \in \rat \Q x,
        \quad 1 \leq i \leq k.
    \end{align}
    %
    Show that $f_1$ is D-finite.
\end{exercise}
