\begin{exercise}\label{ex:inv}
    Let $\K$ be a field of characteristic 0 and let $f \in \powerseries{\K}{x}$.
    \begin{enumerate}
        \item Determine a necessary and sufficient condition for $f$
            to admit a multiplicative inverse.
        \item Does this characterization extend to formal power
            series over a commutative ring? What about invertibility
            with respect to the Hadamard product?
        \item Determine the multiplicative inverses of
            $\sum_{n \in \N} x^n$, $(1-x)^3$ and, more generally,
            $(1-x)^k$ for $k \in \N$.
    \end{enumerate}
\end{exercise}
\begin{solution}
    \begin{enumerate}
        \item[]
        \item $f = \sum_{n \in \N} a_n x^n \in \powerseries{\K}{x}$
            is invertible if and only if $a_0 \neq 0$.

            $\Rightarrow$: If $f$ is invertible, there exist
            $f^{-1} = \sum_{n \in \N} b_n x^n \in
            \powerseries{\K}{x}$ such that $ff^{-1} = 1$.
            By identifying the coefficients of $ff^{-1}$ and 1,
            we obtain that $a_0 b_0 = 1$, thus $a_0 \neq 0$.\\
            Alternatively: by contradiction, if $a_0 = 0$ then $f =
            \sum_{n \geq 1} a_n x^n$ and, for all $g \in
            \powerseries{\K}{x}$, $\mathrm{ord}(fg) =
            \mathrm{ord}(f) + \mathrm{ord}(g) = 1+\mathrm{ord}(g)
            > 0 = \mathrm{ord}(1)$.
            Thus, $f$ cannot be invertible.

            $\Leftarrow$: Conversly, if $a_0 \neq 0$ and since
            $\K$ is a field, $a_0$ is invertible.
            Let us denote its inverse by $a_0^{-1}$ and define
            $f^{-1} = \sum_{n \in \N} b_n x^n \in
            \powerseries{\K}{x}$ inductively as follows.
            By the indentity $ff^{-1} = \sum_{n \in \N} \sum_{i=0}^n
            a_i b_{n-i} x^n = 1$, we necessarily have $a_0 b_0 = 1$
            and thus $b_0 = a_0^{-1}$.
            Moreover, for all $n \geq 1$, $\sum_{i=0}^n a_i b_{n-i} =
            0$ and thus $b_n =a_0^{-1}\left(-\sum_{i=1}^n a_i b_{n-i}\right)$.
            Note that each $b_n$ only depend on the definition of the
            previous $b_i$ for $i<n$.
        \item For formal power series over a commutative ring, the
            characterization becomes: $f \in \powerseries{R}{x}$ is
            invertible if and only if $f(0)$ is invertible.
            The proof remains the same.

            For the Hadamard product, the neutral element is the
            series $\sum_{n \in \N} x^n$ and thus, $f$ is invertible
            if and only if all its coefficients are invertible.
        \item For $\sum_{n \in \N} x^n$, notice that $\sum_{n \in \N}
            x^n(1-x)= \sum_{n \in \N} x^n - \sum_{n \geq 1} x^n = 1$.

            For $(1-x)^3$, by derivating both sides of the equality
            $\frac{1}{1-x} = \sum_{n \in \N} x^n$, we obtain
            $\frac{1}{(1-x)^2} = \sum_{n \in \N} (n+1) x^n$.
            Derivating a second time yields $\frac{2}{(1-x)^3} =
            \sum_{n \in \N} (n+1)(n+2) x^n$, \ie $\frac{1}{(1-x)^3} =
            \sum_{n \in \N} \frac{(n+1)(n+2)}{2} x^n$.

            Generalizing the pattern observed previously, we can show
            by induction that $\frac{1}{(1-x)^k} = \sum_{n \in \N}
            \binom{n+k-1}{n} x^n$.
    \end{enumerate}
\end{solution}

\begin{exercise}
    Determine explicit formulas for the coefficients of
    $\frac{x^2}{(1-x)(1-2x)}$ and $\frac{1+x}{1-2x+x^2}$.
\end{exercise}
\begin{solution}
    For $\frac{x^2}{(1-x)(1-2x)}$, using partial fraction
    decomposition (\ie decomposing $\frac{1}{(1-x)(1-2x)}$ as
    $\frac{a}{1-x}+\frac{b}{1-2x}$ for some $a,b \in \Q$) we obtain:
    \begin{align*}
        \frac{x^2}{(1-x)(1-2x)} &= x^2
        \left(\frac{2}{1-2x}-\frac{1}{1-x}\right) \\
        &= x^2 \left(2\sum_{n \in \N} 2^n x^n-\sum_{n \in \N} x^n\right)  \\
        \frac{x^2}{(1-x)(1-2x)} &= \sum_{n \geq 2} (2^{n-1}-1) x^n
    \end{align*}
    Alternative combinatorial approach: directly compute the Cauchy product of
    $x^2$, $\frac{1}{1-x}$ and $\frac{1}{1-2x}$.

    For $\frac{1+x}{1-2x+x^2}=\frac{1+x}{(1-x)^2}$, we can use
    Exercise~\ref{ex:inv} to obtain $\sum_{n \in \N} (2n+1) x^n$.
\end{solution}

\section*{Rational series and linear recurrence sequence}
The Fibonacci sequence $(F_n)_{n \in \N}$ is defined by $F_0 = 0,
F_1 = 1$ and $F_n = F_{n-1} + F_{n-2}$, for all $n \geq 2$.
\begin{exercise}
    Determine the generating function of $(F_n)_{n \in \N}$.
\end{exercise}
\begin{solution}
    Let $f = \sum_{n \in \N} F_n x^n$ be the generating function of
    the Fibonacci sequence.
    \begin{align*}
        f &= \sum_{n \in \N} F_n x^n
        = F_0 + F_1 x + \sum_{n \geq 2} F_n x^n \\
        &= x + \sum_{n \geq 2} F_{n-1} x^n + \sum_{n \geq 2} F_{n-2} x^n \\
        &= x + x\sum_{n \geq 2} F_{n-1} x^{n-1} + x^2\sum_{n \geq
        2} F_{n-2} x^{n-2} \\
        &= x + x (f- F_0)+ x^2 f \\
        f &= \frac{x}{1-x-x^2} \qedhere
    \end{align*}
\end{solution}

The Fibonacci sequence is a particular case of \emph{linear
recurrence sequences}, \ie sequences $(a_n)_{n \in \N}$ satisfying
a recurrence of the form $a_n = \sum_{i=1}^k c_i a_{n-i}$ for
some constants $c_1,\dots,c_k$.
\begin{exercise}\label{ex:lrs}
    Show that $(a_n)_{n \in \N}$ is a linear recurrence
    sequence if and only if its generating function is rational.
\end{exercise}
\begin{solution}
    Let $f = \sum_{n \in \N} a_n x^n$ be the generating function of
    $(a_n)_{n \in \N}$.

    If $(a_n)_{n \in \N}$ is a linear recurrence sequence then, by
    definition, there exist $N \geq k$ such that, for all $n>N$,
    $a_n = \sum_{i=1}^k c_i a_{n-i}$.

    For all $i \leq N$, let $P_i = \sum_{n=0}^i a_n x^n \in \poly{\K}{x}$.
    Then,
    \begin{align*}
        f &= P_N+ \sum_{n>N} a_n x^n
        = P_N+ \sum_{n>N} \left(\sum_{i=1}^k c_i a_{n-i}\right) x^n \\
        &= P_N+ \sum_{i=1}^k c_i \sum_{n>N} a_{n-i} x^n \\
        &= P_N+ \sum_{i=1}^k c_i x^i \sum_{n>N} a_{n-i} x^{n-i} \\
        &= P_N+ \sum_{i=1}^k c_i x^i (f-P_{N-i}) \\
        f &= \frac{P}{Q} \tag{where $P=P_N- \sum_{i=1}^k c_i x^i
        P_{N-i}$ and $Q=1-\sum_{i=1}^k c_i x^i$}
    \end{align*}
    Thus, $f$ is a rational series.

    Conversely, any rational series can be written in the form of the
    last equality, and running the calculations in reverse yields the
    linear recurrence relation verified by $(a_n)_{n \in \N}$.
\end{solution}

\begin{exercise}
    The \emph{exponential generating function} of a sequence
    $(a_n)_{n \in \N}$ is defined as the formal power series $\sum_{n
    \in \N} a_n \frac{x^n}{n!}$.
    \begin{enumerate}
        \item Show that $(a_n)_{n \in \N}$ is a linear recurrence sequence
            if and only if its exponential generating function satisfies a
            linear differential equation with constant coefficients.
        \item Deduce a closed form for the terms of the Fibonacci
            sequence from the linear differential equation verified
            by its exponential generating function.
    \end{enumerate}
\end{exercise}
\begin{solution}
    \begin{enumerate}
        \item[]
        \item The idea is the same as in Exercise~\ref{ex:lrs}, using
            derivations to shift the coefficients of $f$ instead.

            Let $f = \sum_{n \in \N} a_n \frac{x^n}{n!}$ be the
            exponential generating function of $(a_n)_{n \in \N}$.

            If $(a_n)_{n \in \N}$ is a linear recurrence sequence,
            there exist $k \in \N$ such that, for all $n \in \N$,
            $a_{n+k} = \sum_{i=1}^k c_i a_{n+k-i}$.

            Note that, for all $i \in \N$, $f^{(i)} = \sum_{n \in \N}
            a_{n+i} \frac{x^n}{n!}$.
            Then,
            \begin{align*}
                f^{(k)} &= \sum_{n \in \N} a_{n+k} \frac{x^n}{n!} \\
                &= \sum_{n \in \N} \left(\sum_{i=1}^k c_i
                a_{n+k-i}\right) \frac{x^n}{n!} \\
                &= \sum_{i=1}^k c_i \sum_{n \in \N} a_{n+k-i} \frac{x^n}{n!} \\
                f^{(k)} &= \sum_{i=1}^k c_i f^{(k-i)}
            \end{align*}

            Running the calculations in reverse yields the converse.
        \item Using the previous question, we known that the
            exponential generating function $f = \sum_{n \in \N} F_n
            \frac{x^n}{n!}$ of the Fibonacci sequence verifies the
            linear differential equation $f''- f'-f = 0$.

            The characteristic polynomial of the differential
            equation is $x^2-x-1$ whose roots are $\varphi =
            \frac{1+\sqrt{5}}{2}$ and $\psi = \frac{1-\sqrt{5}}{2}$.
            Its solutions are then of the form $f = a e^{\varphi x} +
            b e^{\psi x}$ for some $a,b \in \Q$.
            Using the initial conditions $f(0) = F_0 = 0$ and $f'(0)
            = F_1 = 1$, we obtain the system of linear equations:
            \[
                \begin{cases}
                    a+b = 0 \\
                    \varphi a + \psi b = 1
                \end{cases}
            \]
            Solving the system, we find that $f =
            \frac{1}{\sqrt{5}}(e^{\varphi x} - e^{\psi x})
            = \sum_{n \in \N} \frac{\varphi^n - \psi^n}{\sqrt{5}} x^n$, and thus
            $F_n = \frac{\varphi^n - \psi^n}{\sqrt{5}}$.
    \end{enumerate}
\end{solution}

Deciding whether a given linear recurrence sequence contains a zero
term is a long standing open problem known as the \emph{Skolem problem}.

In the multivariate setting, the generating function of a linear
recurrence sequence is not necessarily rational~\cite{BouPet2000}.
Moreover, we can show that the following \enquote{analogue} of the Skolem
problem is undecidable.
\begin{exercise}
    Let $k \geq 9$.
    Show that the following decision problem is undecidable:
    given a rational series $f \in \powerseries{\Q}{x_1,\dots,x_k}$,
    does there exists $n_1,\dots,n_k \in \N$ such that
    $\coef{x_1^{n_1},\dots,x_k^{n_k}}{f} = 0$?

    \emph{Hint:} Hilbert's tenth problem (given a polynomial $p \in
        \poly{\Q}{x_1,\dots,x_k}$, does there exists $n_1,\dots,n_k \in
    \N$ such that $p(n_1,\dots,n_k) = 0$) is undecidable for 9 or
    more variables.
\end{exercise}
\begin{solution}
    This result is due to Litow~\cite{Lit2003}.

    To get a reduction from Hilbert's tenth problem, we need to
    define a computable function mapping each polynomial $p \in
    \poly{\Q}{x_1,\dots,x_k}$ to a rational series $f_p \in
    \powerseries{\Q}{x_1,\dots,x_k}$ such that
    $\coef{x_1^{n_1},\dots,x_k^{n_k}}{f} = p(n_1,\dots,n_k)$ for all
    $n_1,\dots,n_k \in \N$.

    We define this function on all the monomials
    $p = x_1^{r_1} \dots x_k^{r_k}$ for all $r_1,\dots,r_k \in \N$
    and the result follows directly from its linear extension since
    both rationality and the desired property are preserved by linear
    combinations.

    Observe that, in the univariate case, applying a derivation
    followed by a multiplication by $x$ to $\frac{1}{1-x}$, we obtain
    the series $\sum_{n \in \N} n x^n$.
    Iterating this operation, which we denote by $d_x$, we can then
    define, for all monomial $p=x^r \in \poly{\Q}{x}$ a rational series $f_p =
    d_x^r \left(\frac{1}{1-x}\right) \in \powerseries{\Q}{x}$ such that
    $\coef{x^n}{f_p} = n^r = p(n)$ for all $n \in \N$.

    Generalizing this idea to the multivariate case, define the
    operations $\partial_i$ by $\partial_i (f) = x_i \frac{\partial
    f}{\partial x_i}$ for all $1 \leq i \leq k$ and $f \in
    \powerseries{\Q}{x_1,\dots,x_k}$.
    We can then define, given any monomial $p = x_1^{r_1} \dots x_k^{r_k}$,
    the rational series $f_p = \partial_1^{r_1} \dots
    \partial_k^{r_k} \left(\frac{1}{(1-x_1)\dots(1-x_k)}\right)$
    which verifies the desired property.
\end{solution}
